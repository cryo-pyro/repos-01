function computeVelocity(history) {
  if (history.length < 2) return null;

  const last = history[history.length - 1].P;
  const prev = history[history.length - 2].P;

  let velocity = {};

  for (let key in last) {
    velocity[key] = last[key] - prev[key];

  return velocity;
}
