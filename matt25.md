# ukb_renal_epsilon_sgd_prototype.py
# Exact code that produced the two plots + the quantified gap analysis
# Run this in Colab / Jupyter / local Python (PyTorch + matplotlib)

import torch
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

# ====================== SYNTHETIC DATA (your v0 public proxies) ======================
np.random.seed(42)
torch.manual_seed(42)

N_DISTRICTS = 146
N_OBSERVED = 11

# Realistic population distribution (Uganda 2024 approx)
population = np.random.lognormal(mean=12.65, sigma=0.6, size=N_DISTRICTS)  # ~300k average
population = (population / population.sum() * 45_905_417).astype(int)  # national total

# Facilities proxy (urban bias: observed districts have more)
facilities = np.random.poisson(lam=15, size=N_DISTRICTS)
facilities[:N_OBSERVED] += np.random.poisson(lam=25, size=N_OBSERVED)  # urban boost

# Normalised features for the model
X = np.column_stack([population, facilities]).astype(np.float32)
X = (X - X.mean(axis=0)) / X.std(axis=0)  # standardise

# ε = 1 for the 135 blank districts (missing data signal)
epsilon = np.ones(N_DISTRICTS, dtype=np.float32)
epsilon[:N_OBSERVED] = 0.0

# Ground-truth proxy burden (higher in blanks because of real inequity)
true_burden = 0.0008 * population + 0.3 * facilities + np.random.normal(0, 5, N_DISTRICTS)
true_burden = np.clip(true_burden, 5, 55).astype(np.float32)

# Observed targets (only the 11 green districts)
y_obs = true_burden[:N_OBSERVED].copy()

# ====================== PYTORCH MODEL + ε-WEIGHTED SGD ======================
class BurdenPredictor(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(2, 1)  # pop + facilities → predicted burden
    
    def forward(self, x):
        return self.linear(x).squeeze(-1)

model = BurdenPredictor()
optimizer = torch.optim.SGD(model.parameters(), lr=0.05, momentum=0.9)
criterion = torch.nn.MSELoss()

X_tensor = torch.from_numpy(X)
y_obs_tensor = torch.from_numpy(y_obs)

# Training loop: normal loss on observed + λ·ε·(pred - proxy)^2 on ALL
lambda_epsilon = 0.8
epochs = 1200

for epoch in range(epochs):
    optimizer.zero_grad()
    
    pred = model(X_tensor)
    
    # Loss on the 11 observed districts
    loss_obs = criterion(pred[:N_OBSERVED], y_obs_tensor)
    
    # ε-weighted penalty on EVERY district (this is the "descending the voids" part)
    loss_epsilon = lambda_epsilon * (epsilon * (pred - torch.from_numpy(true_burden))**2).mean()
    
    loss = loss_obs + loss_epsilon
    loss.backward()
    optimizer.step()
    
    if epoch % 300 == 0:
        print(f"Epoch {epoch} | Total loss: {loss.item():.3f}")

# Final predictions after SGD has felt the pull of the blanks
pred_final = model(X_tensor).detach().numpy()

# ====================== QUANTIFICATION (the brutal numbers) ======================
pop_covered = population[:N_OBSERVED].sum()
pop_blank = population.sum() - pop_covered
print(f"Population covered by published data: {pop_covered/1e6:.1f}M ({pop_covered/population.sum()*100:.1f}%)")
print(f"Population in 135 blanks: {pop_blank/1e6:.1f}M ({pop_blank/population.sum()*100:.1f}%)")

# ====================== VISUALISATION 1: TOP 20 GRAVITY WELLS ======================
epsilon_weighted_burden = epsilon * pred_final
top20_idx = np.argsort(epsilon_weighted_burden)[-20:][::-1]

plt.figure(figsize=(12, 6))
bars = plt.bar(range(20), epsilon_weighted_burden[top20_idx], 
               color=sns.color_palette("Reds", 20))
plt.title('Top 20 Districts by ε-weighted Burden (the "gravity wells" SGD is now descending)')
plt.xlabel('District ID (synthetic)')
plt.ylabel('ε × Predicted Kidney Burden')
plt.xticks(range(20), top20_idx, rotation=45)
plt.grid(axis='y', alpha=0.3)
plt.tight_layout()
plt.savefig('ukb_renal_gap_bars.png', dpi=300, bbox_inches='tight')
plt.show()

# ====================== VISUALISATION 2: SCATTER (the fractal manifold view) ======================
plt.figure(figsize=(10, 7))
scatter = plt.scatter(population, pred_final, 
                      c=epsilon, cmap='RdYlGn_r', s=80, alpha=0.85,
                      edgecolors='k', linewidth=0.5)

plt.colorbar(scatter, label='ε (Missingness)', ticks=[0, 1])
plt.title('Predicted Burden vs Population\n(green = data districts, red = ε=1 blanks)')
plt.xlabel('Population')
plt.ylabel('Predicted Burden')
plt.grid(alpha=0.3)
plt.tight_layout()
plt.savefig('ukb_renal_scatter.png', dpi=300, bbox_inches='tight')
plt.show()

print("✅ Plots saved as ukb_renal_gap_bars.png and ukb_renal_scatter.png")
print("✅ ε-SGD loop complete — the blanks are now inside the optimisation.")
