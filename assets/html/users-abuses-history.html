🎲 Transition Function (Core Engine)

sql
CREATE OR REPLACE FUNCTION simulate_ckd_progression(p_id UUID, steps INT)
RETURNS VOID AS $$
DECLARE
    t INT := 0;
    current_state TEXT := 'RISK';
    current_egfr NUMERIC := 95;

    bp NUMERIC;
    sugar NUMERIC;

    epsilon NUMERIC;
    shock BOOLEAN;

BEGIN
    -- pull baseline risk
    SELECT systolic_bp, hba1c INTO bp, sugar
    FROM risk_factors
    WHERE patient_id = p_id
    ORDER BY recorded_at DESC
    LIMIT 1;

    WHILE t < steps LOOP

        -- ε: heavy-tailed noise
        epsilon := (random() - 0.5) * 10;

        -- rare shock event (Macduff)
        shock := random() < 0.05;

        IF shock THEN
            epsilon := epsilon * 5;
        END IF;

        -- disease progression (decline in eGFR)
        current_egfr := current_egfr 
            - (0.5 + 0.01*bp + 0.02*sugar + epsilon);

        -- state transitions
        IF current_egfr > 60 THEN
            current_state := 'CKD_1_2';
        ELSIF current_egfr > 30 THEN
            current_state := 'CKD_3_4';
        ELSIF current_egfr > 15 THEN
            current_state := 'ESRD';
        ELSIF current_egfr > 10 THEN
            current_state := 'DIALYSIS';
        ELSE
            current_state := 'TRANSPLANT_OR_DEATH';
        END IF;

        INSERT INTO patient_simulation(patient_id, state, egfr, time_step)
        VALUES (p_id, current_state, current_egfr, t);

        t := t + 1;
    END LOOP;
END;
$$ LANGUAGE plpgsql;
