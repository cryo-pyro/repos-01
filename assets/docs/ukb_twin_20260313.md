```python
class PentadOptimizer:
    """
    The engine for the Ukubona-Pentad-v2.0.
    Treats the LLM as the 'Gradient Computer' to update the Digital Twin (theta).
    """
    
    def __init__(self, model_name: str = "gemini-1.5-pro"):

        Calculate 'eta' (Learning Rate) based on evidence precision (0.0 to 1.0).
        Return the updated JSON only.
        """

        Calculates the delta and updates theta_t to theta_t1.
```

```python
from fastapi import FastAPI, UploadFile, File, Form
from fastapi.middleware.cors import CORSMiddleware
import json
import os
from llm_client import PentadOptimizer

app = FastAPI(title="Ukubona Learning Twin API")

# Enable CORS for local development (VS Code Live Server)
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

optimizer = PentadOptimizer()

@app.post("/update-twin/{twin_id}")
async def update_twin(
    twin_id: str,
    user_input: str = Form(...),
    file: UploadFile = File(None)
):
    # 1. Load the current state theta_t
    file_path = f"assets/json/pentad_{twin_id}.json"
    if not os.path.exists(file_path):
        return {"error": "Twin not found"}
    
    with open(file_path, "r") as f:
        current_theta = json.load(f)

    # 2. Save uploaded image temporarily if it exists
    temp_path = None
    if file:
        temp_path = f"temp_{file.filename}"
        with open(temp_path, "wb") as buffer:
            buffer.write(await file.read())

    # 3. Compute the Gradient and Update (The LLM Step)
    new_theta = optimizer.calculate_update(current_theta, user_input, temp_path)

    # 4. Persistence: Write theta_t+1 to disk
    with open(file_path, "w") as f:
        json.dump(new_theta, f, indent=2)

    # Clean up temp file
    if temp_path and os.path.exists(temp_path):
        os.remove(temp_path)

    return new_theta

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
```

# 5. Minimal Python implementation

Here is a **clean prototype** of your learning twin update.

```python
import numpy as np

def normalize(state):
    total = sum(state.values())
    return {k: v / total for k, v in state.items()}


def update_state(state, gradient, eta):
    
    new_state = {}

    for diagnosis in state:

        delta = gradient.get(diagnosis, 0)

        new_state[diagnosis] = state[diagnosis] + eta * delta

        if new_state[diagnosis] < 0:
            new_state[diagnosis] = 0

    return normalize(new_state)
```

```html
  if (t.includes('hive') || t.includes('wheal') || t.includes('welt') || t.includes('rash')) { g.urticaria += 0.30; g.aquagenic_pruritus -= 0.15; }
  if (t.includes('no rash') || t.includes('no hive') || t.includes('clear skin')) { g.urticaria -= 0.20; g.aquagenic_pruritus += 0.10; }
  if (t.includes('dry') || t.includes('flak') || t.includes('crack')) g.xerosis += 0.25;
  if (t.includes('jak2') || t.includes('hematocrit') || t.includes('polycyth') || t.includes('blood count')) g.polycythemia_vera += 0.40;
```

```html
function normalize(s) {
  const total = Object.values(s).reduce((a, b) => a + b, 0);
  const out = {};
  for (const k in s) out[k] = Math.max(0, s[k]) / total;
  return out;
}
```

```html
}

function toBase64(file) {
  return new Promise((res, rej) => {
    const r = new FileReader();
    r.onload = () => res(r.result.split(',')[1]);

}

async function callClaude() {
  let messages;

  if (currentTab === 'text') {
    const text = document.getElementById('obs-input').value.trim();
    if (!text) throw new Error('no observation entered');
    messages = [{ role: 'user', content: text }];

    ];

  } else if (currentTab === 'pdf') {

    const ctx = document.getElementById('pdf-context').value.trim();
    const content = [
      { type: 'document', source: { type: 'base64', media_type: 'application/pdf', data: currentPdfData } },
      { type: 'text', text: ctx || 'Extract all clinically relevant findings and compute the diagnostic gradient.' }
    ];
    messages = [{ role: 'user', content }];
  }

  const response = await fetch('https://api.anthropic.com/v1/messages', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      model: 'claude-sonnet-4-20250514',
      max_tokens: 1000,
      system: SYSTEM_PROMPT,
      messages
    })
  });

  const data = await response.json();
  if (data.error) throw new Error(data.error.message);
  const text = data.content.filter(b => b.type === 'text').map(b => b.text).join('');
  return JSON.parse(text.replace(/```json|```/g, '').trim());
}
```

```html
function normalize(s) {
  const total = Object.values(s).reduce((a,b) => a+b, 0);
  const out = {};
  for (const k in s) out[k] = Math.max(0, s[k]) / total;
  return out;
}
```

```html
function addTrace(result,eta){
  document.getElementById('trace-empty').style.display='none';
  const shifts=Object.entries(result.gradient).filter(([,v])=>Math.abs(v)>0.01).sort((a,b)=>Math.abs(b[1])-Math.abs(a[1])).map(([k,v])=>`${v>0?'↑':'↓'} ${DIAGNOSES[k].label} (${v>0?'+':''}${v.toFixed(2)})`).join(' · ');
  const line=document.createElement('div');
  line.style.cssText='border-bottom:0.5px solid var(--color-border-tertiary);padding-bottom:6px;margin-bottom:6px;';

}

function setLoading(loading){const btn=document.getElementById('compute-btn');btn.textContent=loading?'⟳ computing...':'∇ compute gradient';btn.disabled=loading;}

async function pushUpdate(){
  const eta=parseFloat(document.getElementById('modality').value);
  setLoading(true);
  try {
    const result=await callClaude();
    state=applyGradient(result.gradient,eta);
    saveVersion(result,eta);
    addTrace(result,eta);
    renderBars(); renderMetrics();
    evoChart.data.labels.push('t'+updateCount);
    evoChart.data.datasets.forEach((ds,i)=>{ds.data.push(Object.values(state)[i]);});

  }
  setLoading(false);
```

A single `index.html` is all you need. No backend, no server, no directory structure. The API call goes directly from the browser to Anthropic — it's fully client-side.One file. Deploy it anywhere in 60 seconds.

**The API key flow** — the key never leaves the browser. It's stored in `localStorage` under `ukb_api_key`. The top bar has a password input — paste your key once, hit save, it persists across visits. No backend, no server ever sees it.

**One important note:** the file uses `anthropic-dangerous-direct-browser-access: true` in the headers — that's the correct flag for direct browser-to-API calls. It works, but means your API key is client-side. Fine for personal use, not for sharing publicly. If you ever make it multi-user, you'll want a thin proxy in the middle.
